cmake_minimum_required(VERSION 3.20)
project(EngineReference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(ROOT "${ENGINE_REFERENCE_SOURCE}" CACHE PATH "Snapshot original adapté")
if(NOT EXISTS "${ROOT}/src/synthesizer.cpp")
    message(FATAL_ERROR "Utiliser Build.py pour préparer la référence")
endif()
set(PIR ${ROOT}/dependencies/submodules/piranha)
find_package(FLEX REQUIRED)
find_package(BISON 3.2 REQUIRED)
FLEX_TARGET(lexer ${PIR}/flex-bison/scanner.l ${CMAKE_CURRENT_BINARY_DIR}/scanner.auto.cpp DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/scanner.auto.h)
BISON_TARGET(parser ${PIR}/flex-bison/specification.y ${CMAKE_CURRENT_BINARY_DIR}/parser.auto.cpp DEFINES_FILE ${CMAKE_CURRENT_BINARY_DIR}/parser.auto.h)
ADD_FLEX_BISON_DEPENDENCY(lexer parser)
file(GLOB PIR_SOURCES ${PIR}/src/*.cpp)
file(GLOB SCS_SOURCES ${ROOT}/dependencies/submodules/simple-2d-constraint-solver/src/*.cpp)
file(GLOB SCRIPT_SOURCES ${ROOT}/scripting/src/*.cpp)
file(READ ${ROOT}/CMakeLists.txt ORIGINAL_CMAKE)
string(REGEX MATCH "add_library\\(engine-sim STATIC([^)]*)\\)" CORE_LIST "${ORIGINAL_CMAKE}")
string(REGEX MATCHALL "src/[a-z_]+\\.cpp" CORE_SOURCES "${CORE_LIST}")
list(TRANSFORM CORE_SOURCES PREPEND "${ROOT}/")
add_library(reference STATIC ${CORE_SOURCES} ${SCS_SOURCES} ${PIR_SOURCES} ${SCRIPT_SOURCES} ${BISON_parser_OUTPUTS} ${FLEX_lexer_OUTPUTS})
target_include_directories(reference PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(reference PUBLIC ${ROOT}/dependencies/submodules ${ROOT}/include ${ROOT}/scripting/include ${PIR}/include ${ENGINE_REFERENCE_FLEX_INCLUDE} ${CMAKE_CURRENT_BINARY_DIR})
target_compile_options(reference PUBLIC -fms-extensions -D__forceinline=inline -include ${CMAKE_CURRENT_SOURCE_DIR}/Portability.hpp -Wno-deprecated-declarations -Wno-switch -Wno-unused-value -Wno-delete-non-abstract-non-virtual-dtor)
add_executable(RenderReference Main.cpp Drive.cpp)
target_link_libraries(RenderReference reference)
target_include_directories(RenderReference PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ENGINE_REFERENCE_GENERATED})

add_executable(CycleCheck CycleCheck.cpp)
target_link_libraries(CycleCheck reference)
target_include_directories(CycleCheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
enable_testing()
add_test(NAME ignition-cycle COMMAND CycleCheck)

option(ENGINE_REFERENCE_VECTOR_FIR "Autoriser la réassociation dans la somme FIR uniquement" ON)
if(ENGINE_REFERENCE_VECTOR_FIR)
    set_source_files_properties(${ROOT}/src/convolution_filter.cpp PROPERTIES COMPILE_OPTIONS "-ffast-math")
endif()

add_executable(ExhaustBlendCheck ExhaustBlendCheck.cpp)
target_include_directories(ExhaustBlendCheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME exhaust-blend COMMAND ExhaustBlendCheck)

add_executable(CalculatedExhaustCheck CalculatedExhaustCheck.cpp)
target_include_directories(CalculatedExhaustCheck PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME calculated-exhaust COMMAND CalculatedExhaustCheck)
